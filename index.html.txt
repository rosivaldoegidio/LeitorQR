<script>

const URL_API = "https://script.google.com/macros/s/AKfycbzaPKDfmkvKfBjANAulitSsJQaxDTzuzwRx0Vl147be4Z33V-CDtv8zRuo7iOx2JKpL/exec";

async function enviar(){

    const codigo =
    document.getElementById("codigo").value;

    const resposta = await fetch(URL_API,{
        method:"POST",
        headers:{
            "Content-Type":"application/json"
        },
        body:JSON.stringify({
            codigo:codigo
        })
    });

    const json = await resposta.json();

    document.getElementById("resultado").innerHTML =
    JSON.stringify(json);

}

</script>
